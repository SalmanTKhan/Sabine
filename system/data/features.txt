// Sabine
// Database File
// 
// A hierarchical feature tree that allows toggling entire feature sets
// on and off. It's geared towards enabling certain updates in one go,
// but it's also possible to pick and choose certain features.
// See docs/features.txt for more information.
// 
// Note that some features in this file are currently for reference only,
// and don't necessarily change the behavior of the server.
//---------------------------------------------------------------------------

[
{ name: "Alpha", enabled: true, children: [
	// Enables all first class jobs. At one point, players could only play
	// as Swordmen.
	{ name: "FirstClassJobs", enabled: true },
]},

{ name: "Beta1", enabled: false, children: [
	// Enables characters to raise their job level.
	{ name: "JobLevels", enabled: true },

	// Enables the mage class, which was not available in the alpha.
	{ name: "Mage", enabled: true },

	// Enables skills, which were not available in the alpha.
	// This feature doesn't do anything if the alpha client is used.
	{ name: "Skills", enabled: true },

	// Gives characters a higher max weight.
	{ name: "HigherMaxWeight", enabled: true },
]},

// Selects rAthena PRE-RE behavior over eAthena classic. Affects
// data file selection (e.g. produce_db) and a handful of caps that
// rAthena raised pre-renewal. Has no effect when "Renewal" is on.
{ name: "PreRenewal", enabled: false },

// Switches damage calculation to the renewal formulas (multiplicative
// DEF/MDEF, weapon ATK split into mastery and base ATK, etc.). Disabled
// by default; classic/pre-renewal formulas are used while this is off.
{ name: "Renewal", enabled: false },

// Master toggle for the expanded status-effect catalogue (Curse and
// future additions). Disable to fall back to the originally-shipped
// SC bundle.
{ name: "StatusEffectsExpanded", enabled: true },

// When enabled, status-resistance durations follow the rAthena renewal
// formula (status_get_sc_def #ifdef RENEWAL branch). Disabled by default
// so classic VIT/LUK math is used.
{ name: "RenewalStatusResistance", enabled: false },

// When enabled, base SC durations follow rAthena db/re/status.yml; when
// disabled they follow db/pre-re/status.yml. Independent of the Renewal
// damage flag so operators can mix-and-match.
{ name: "RenewalStatusDurations", enabled: false },
]
